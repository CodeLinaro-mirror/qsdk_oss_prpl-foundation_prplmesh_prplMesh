find_package(Doxygen)
if(NOT DOXYGEN_FOUND)
    message(STATUS "Doxygen not found -- 'doxygen' target will not be available")
    return()
endif()

set(DOXYGEN_PROJECT_NAME    "prplMesh")
set(DOXYGEN_PROJECT_BRIEF   "Multi-AP EasyMesh controller and agent")
set(DOXYGEN_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/docs")
set(DOXYGEN_USE_MDFILE_AS_MAINPAGE "${CMAKE_SOURCE_DIR}/README.md")

set(DOXYGEN_RECURSIVE   YES)
set(DOXYGEN_EXTRACT_ALL YES)

# Cleaner output
set(DOXYGEN_GENERATE_TREEVIEW   YES)
set(DOXYGEN_JAVADOC_AUTOBRIEF   NO)
set(DOXYGEN_QT_AUTOBRIEF        NO)
set(DOXYGEN_MULTILINE_CPP_IS_BRIEF NO)

# Exclude generated / third-party files
set(DOXYGEN_EXCLUDE_PATTERNS
    "*/build/*"
    "*/tlvf/AutoGenerated/*"
    "*_test*"
    "*_tests*"
    "*/tests/*"
    "*/test/*"
)

doxygen_add_docs(doxygen
    "${CMAKE_SOURCE_DIR}/framework"
    "${CMAKE_SOURCE_DIR}/common"
    "${CMAKE_SOURCE_DIR}/controller"
    "${CMAKE_SOURCE_DIR}/agent"
    COMMENT "Generating API documentation with Doxygen"
)
